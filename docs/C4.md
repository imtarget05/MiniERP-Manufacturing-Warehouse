## C4-Context

```mermaid
C4Context
    title MiniERP — System Context
    Person(wh, "Warehouse / planner", "receive, issue, produce, adjust")
    Person(aud, "Auditor / manager", "approvals, genealogy, reports")
    System(erp, "MiniERP", "Oracle Tx core + .NET API + dashboard")
    System_Ext(hd, "Helpdesk Lab", "fail-soft incident receiver")
    System_Ext(zeb, "Zebra printers", "ZPL labels")
    Rel(wh, erp, "operates", "HTTPS/API")
    Rel(aud, erp, "approves, traces", "dashboard")
    Rel(erp, hd, "outbox posts", "HTTP")
    Rel(erp, zeb, "prints", "ZPL")
```

## C4-Container

```mermaid
C4Container
    title MiniERP — Containers
    Person(wh, "Warehouse", "dashboard/API")
    Container_Boundary(e, "MiniERP") {
        Container(api, "ASP.NET Core 8 API", ".NET", "62 ops, RBAC, X-Correlation-ID")
        Container(dash, "Operations dashboard", "web", "stock, BOM, lots, approvals")
        ContainerDb(ora, "Oracle 23c", "PL/SQL", "31 tables, 3 packages, FEFO, ledger")
        Container(out, "Helpdesk outbox", "jobs", "fail-soft retry posts")
    }
    System_Ext(hd, "Helpdesk", "receiver")
    Rel(wh, api, "calls", "HTTPS")
    Rel(wh, dash, "views", "HTTPS")
    Rel(api, ora, "transactions", "SQL")
    Rel(out, hd, "posts", "HTTP")
```
